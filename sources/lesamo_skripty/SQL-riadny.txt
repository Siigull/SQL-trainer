-- a) Kolik zboží z kategorie „Monitory“ bylo zakoupeno v jednotlivých měsících roku 2019? (mesic, pocet_nakupu)
SELECT TO_CHAR(datum, 'MM') mesiac, COUNT(*) pocet_nakupu FROM Koupil NATURAL JOIN Zbozi
    WHERE kategorie = 'monitory' AND datum BETWEEN TO_DATE('2019-01-01', 'YYYY-MM-DD') AND TO_DATE('2019-12-31', 'YYYY-MM-DD') GROUP BY TO_CHAR(datum, 'MM');

-- b) Která zboží (kodZb, nazev, kategorie) nakupovali pouze zákazníci narození po roce 2000?
-- chcem vybrat to zbozi kde neexistuje zaznam v koupil kde si to zbozi kupuje clovek narodeny pred rokom 2000
SELECT kodZb, nazev, kategorie FROM Zbozi WHERE NOT EXISTS (SELECT * FROM Zakaznik NATURAL JOIN Koupil WHERE kodZb = Zbozi.kodZb AND dat_nar <= TO_DATE('2000-01-01', 'YYYY-MM-DD'));
    
-- c) Kteří zákazníci (cisloZ, jmeno, mesto) koupili zboží s názvem „Samsung A20“ za nejvyšší cenu? (může být i více zákazníků, kteří zboží koupili za stejnou nejvyšší cenu)
SELECT cisloZ, jmeno, mesto FROM Zakaznik NATURAL JOIN Zbozi NATURAL JOIN Koupil WHERE nazev = 'Samsung A20' AND cena = (SELECT MAX(cena) FROM Zbozi NATURAL JOIN Koupil WHERE nazev = 'Samsung A20');
