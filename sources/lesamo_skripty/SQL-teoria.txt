DROP TABLE Zamestnanci;
DROP TABLE Oddelenia;

CREATE TABLE Oddelenia (
    nazov VARCHAR(50) PRIMARY KEY,
    popis VARCHAR(100)
);

CREATE TABLE Zamestnanci (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    mzda INT NOT NULL CHECK(mzda >= 0),
    meno VARCHAR(50) NOT NULL,
    oddelenie VARCHAR(50) NOT NULL,
    vedeneOddelenie VARCHAR(50) NULL REFERENCES Oddelenia,
    
    FOREIGN KEY (oddelenie) REFERENCES Oddelenia(nazov) ON DELETE CASCADE
);
